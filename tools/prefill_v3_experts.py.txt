# Token counts at or above this run the experts as dequantized BF16 grouped GEMMs (v2); below, Marlin.
GROUPED_MIN_TOKENS = 8192
# torch._grouped_mm (BF16, per-group row offsets) is a private op: present in torch >= 2.8 with a CUDA sm_90 kernel.
# Without it (or off sm_90) every token count stays on Marlin (review of #16).
_HAS_GROUPED_MM = callable(getattr(torch, "_grouped_mm", None))


def _grouped_ok(device):
    return _HAS_GROUPED_MM and torch.cuda.get_device_capability(device) == (9, 0)


@triton.jit
def _nvfp4_dequant_kernel(q_ptr, sf_ptr, out_ptr, ROW_BYTES: tl.constexpr):
    # uint8 [E, N, K/2] (element 2i in the low nibble) with e4m3 [E, N, K/16] -> BF16 [E, N, K] of e2m1 x e4m3, which
    # BF16 holds exactly (at most 5 significant bits); the FP32 tensor scale is applied after the GEMMs (v3, review
    # of #16 m1). One program per weight row.
    row = tl.program_id(0).to(tl.int64)
    c = tl.arange(0, ROW_BYTES)
    b = tl.load(q_ptr + row * ROW_BYTES + c).to(tl.int32)
    sfb = tl.load(sf_ptr + row * (ROW_BYTES // 8) + c // 8)  # e4m3 scale bytes as uint8
    scale = sfb.to(tl.float8e4nv, bitcast=True).to(tl.float32)
    lo = b & 0xF
    hi = (b >> 4) & 0xF
    mlo = lo & 7
    mhi = hi & 7
    # E2M1 magnitudes 0, 0.5, 1, 1.5, 2, 3, 4, 6 for codes 0..7
    vlo = tl.where(mlo < 4, mlo.to(tl.float32) * 0.5, tl.where(mlo < 6, (mlo - 2).to(tl.float32),
                   tl.where(mlo == 6, 4.0, 6.0)))
    vhi = tl.where(mhi < 4, mhi.to(tl.float32) * 0.5, tl.where(mhi < 6, (mhi - 2).to(tl.float32),
                   tl.where(mhi == 6, 4.0, 6.0)))
    vlo = tl.where((lo & 8) != 0, -vlo, vlo) * scale
    vhi = tl.where((hi & 8) != 0, -vhi, vhi) * scale
    v = tl.reshape(tl.join(vlo, vhi), (2 * ROW_BYTES,))  # element 2i = low nibble, 2i + 1 = high
    tl.store(out_ptr + row * (2 * ROW_BYTES) + tl.arange(0, 2 * ROW_BYTES), v.to(tl.bfloat16))


def _dequant(q, sf):
    e, n, kb = q.shape
    out = torch.empty((e, n, 2 * kb), device=q.device, dtype=torch.bfloat16)
    _nvfp4_dequant_kernel[(e * n,)](q, sf.view(torch.uint8), out, ROW_BYTES=kb, num_warps=8)
    return out


@triton.jit
def _gather_local_kernel(x_ptr, perm_ptr, offs_ptr, out_ptr, TOPK: tl.constexpr, NE: tl.constexpr,
                         H: tl.constexpr):
    # row r of the expert-sorted activations = x[perm[r] // TOPK]; only the local routes (r < offs[NE - 1]) are copied
    r = tl.program_id(0)
    if r < tl.load(offs_ptr + NE - 1):
        src = tl.load(perm_ptr + r) // TOPK
        c = tl.arange(0, H)
        tl.store(out_ptr + r.to(tl.int64) * H + c, tl.load(x_ptr + src * H + c))


@triton.jit
def _silu_mul_gs_kernel(h_ptr, out_ptr, eid_ptr, gs_ptr, I: tl.constexpr):
    # local rows only: silu(gate * gs) * (up * gs) in FP32 from the BF16 GEMM-1 row, as silu_and_mul, with the
    # expert's FP32 tensor scale applied here instead of in the weights
    r = tl.program_id(0)
    e = tl.load(eid_ptr + r)
    if e >= 0:
        gs = tl.load(gs_ptr + e)
        c = tl.arange(0, I)
        base = h_ptr + r.to(tl.int64) * (2 * I)
        g = tl.load(base + c).to(tl.float32) * gs
        u = tl.load(base + I + c).to(tl.float32) * gs
        tl.store(out_ptr + r.to(tl.int64) * I + c, (g * tl.sigmoid(g) * u).to(tl.bfloat16))


@triton.jit
def _unsort_gs_kernel(ys_ptr, inv_ptr, lid_ptr, gs_ptr, y_ptr, H: tl.constexpr):
    # slot j = (token, k) of a local route: y[j] = ys[inv[j]] * gs (FP32 multiply, one BF16 rounding); non-local slots
    # are left unwritten (the finalize skips expert id -1, as on the Marlin path)
    j = tl.program_id(0)
    e = tl.load(lid_ptr + j)
    if e >= 0:
        c = tl.arange(0, H)
        src = tl.load(inv_ptr + j)
        v = tl.load(ys_ptr + src * H + c).to(tl.float32) * tl.load(gs_ptr + e)
        tl.store(y_ptr + j.to(tl.int64) * H + c, v.to(tl.bfloat16))


def _experts_grouped(x, topk_weights, topk_ids, state):
    """The routed experts as BF16 grouped GEMMs on per-call dequantized weights (see the docstring).
    Buffers hold T * TOP_K rows: graph capture needs static sizes, and a token's 8 experts can all be local, so that is
    the bound on local routes. Only the local rows are gathered, activated and un-sorted (v3); the GEMMs stop at the
    last local row (offs)."""
    bank, raw = state["bank"], state["raw"]
    num_tokens, size_k = x.shape
    size_n = bank["size_n"]
    x = x.contiguous()
    local_ids = bank["expert_map"][topk_ids].contiguous()  # [T, 8], -1 where not on this GPU
    topk_weights = topk_weights.to(torch.float32).contiguous()
    flat = local_ids.view(-1)
    key = torch.where(flat >= 0, flat, LOCAL_EXPERTS)
    skey, perm = torch.sort(key, stable=True)
    counts = torch.zeros((LOCAL_EXPERTS + 1,), device=x.device, dtype=torch.int32)
    counts.scatter_add_(0, key, torch.ones_like(key, dtype=torch.int32))
    offs = torch.cumsum(counts[:LOCAL_EXPERTS], 0, dtype=torch.int32)
    rows = flat.numel()
    xs = torch.empty((rows, size_k), device=x.device, dtype=x.dtype)
    _gather_local_kernel[(rows,)](x, perm, offs, xs, TOPK=TOP_K, NE=LOCAL_EXPERTS, H=size_k, num_warps=4)
    w13 = _dequant(raw["w13"], raw["w13_sf"])  # [32, 2I, H], without the tensor scale
    h1 = torch._grouped_mm(xs, w13.transpose(1, 2), offs=offs)  # [T*8, 2I]; rows past offs[-1] unwritten
    del w13, xs
    sorted_eid = torch.where(skey < LOCAL_EXPERTS, skey, -1).to(torch.int32)
    h2 = torch.empty((rows, size_n), device=x.device, dtype=x.dtype)
    _silu_mul_gs_kernel[(rows,)](h1, h2, sorted_eid, raw["w13_gs"], I=size_n, num_warps=4)
    del h1
    w2 = _dequant(raw["w2"], raw["w2_sf"])  # [32, H, I]
    ys = torch._grouped_mm(h2, w2.transpose(1, 2), offs=offs)  # [T*8, H]
    del w2, h2
    inv = torch.empty_like(perm)
    inv[perm] = torch.arange(rows, device=x.device, dtype=perm.dtype)
    y = torch.empty((rows, size_k), device=x.device, dtype=x.dtype)
    _unsort_gs_kernel[(rows,)](ys, inv, flat.to(torch.int32), raw["w2_gs"], y, H=size_k, num_warps=4)
    return fused_post_expert_norm_weighted_sum(
        y.view(num_tokens, TOP_K, size_k), topk_weights, norm_weight=state["post_expert_w"], eps=POST_NORM_EPS,
        weight_first=True, slot_expert_ids=local_ids)
